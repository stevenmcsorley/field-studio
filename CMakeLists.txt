cmake_minimum_required(VERSION 3.22)
project(FieldStudio VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FIELD_BUILD_PLUGIN "Build standalone and VST3" ON)
include(CTest)
find_package(Threads REQUIRED)
add_library(field_engine STATIC Source/Engine.cpp Source/TapeAudio.cpp)
target_include_directories(field_engine PUBLIC Source)
set_target_properties(field_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(field_tape_storage STATIC Source/TapeStorage.cpp)
target_include_directories(field_tape_storage PUBLIC Source)
set_target_properties(field_tape_storage PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(field_engine PUBLIC field_tape_storage)
if(BUILD_TESTING)
  add_executable(field_engine_tests tests/engine_tests.cpp)
  target_link_libraries(field_engine_tests PRIVATE field_engine Threads::Threads)
  if(MSVC)
    target_link_options(field_engine_tests PRIVATE /STACK:8388608)
  endif()
  add_test(NAME engine COMMAND field_engine_tests)
  add_executable(field_rt_queue_tests tests/rt_queue_tests.cpp)
  target_include_directories(field_rt_queue_tests PRIVATE Source)
  target_link_libraries(field_rt_queue_tests PRIVATE Threads::Threads)
  add_test(NAME rt_queue COMMAND field_rt_queue_tests)
  add_executable(field_tape_storage_tests tests/tape_storage_tests.cpp)
  target_link_libraries(field_tape_storage_tests PRIVATE field_tape_storage Threads::Threads)
  add_test(NAME tape_storage COMMAND field_tape_storage_tests)
endif()
if(FIELD_BUILD_PLUGIN)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE/CMakeLists.txt")
    set(FIELD_JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE")
    add_subdirectory(vendor/JUCE EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG ea77444750de45e036e0279bb0291144bf8e3688 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
    set(FIELD_JUCE_ROOT "${juce_SOURCE_DIR}")
  endif()
  set(FIELD_FORMATS Standalone VST3)
  if(APPLE)
    list(APPEND FIELD_FORMATS AU)
  endif()
  juce_add_plugin(FieldStudio
    COMPANY_NAME "Field Studio Project"
    BUNDLE_ID "studio.field.instrument"
    PLUGIN_MANUFACTURER_CODE Fstd
    PLUGIN_CODE Fld1
    PRODUCT_NAME "Field Studio"
    FORMATS ${FIELD_FORMATS}
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT TRUE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth Sampler
    NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE)
  target_sources(FieldStudio PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/MidiMapping.cpp Source/MidiSupport.cpp Source/PresetFormat.cpp Source/PresetMapping.cpp Source/PresetParameters.cpp)
  target_include_directories(FieldStudio PUBLIC Source)
  target_compile_definitions(FieldStudio PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
  target_link_libraries(FieldStudio PRIVATE field_engine juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
  if(BUILD_TESTING)
    juce_add_console_app(field_plugin_tests PRODUCT_NAME "Field Studio integration tests")
    target_sources(field_plugin_tests PRIVATE tests/plugin_tests.cpp)
    target_link_libraries(field_plugin_tests PRIVATE FieldStudio)
    target_include_directories(field_plugin_tests PRIVATE $<TARGET_PROPERTY:juce::juce_core,INTERFACE_INCLUDE_DIRECTORIES>)
    target_compile_definitions(field_plugin_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    add_test(NAME plugin COMMAND field_plugin_tests)
    juce_add_console_app(field_midi_tests PRODUCT_NAME "MIDI routing and clock tests")
    target_sources(field_midi_tests PRIVATE tests/midi_tests.cpp)
    target_link_libraries(field_midi_tests PRIVATE FieldStudio)
    target_include_directories(field_midi_tests PRIVATE $<TARGET_PROPERTY:juce::juce_core,INTERFACE_INCLUDE_DIRECTORIES>)
    target_compile_definitions(field_midi_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    add_test(NAME midi COMMAND field_midi_tests)
    juce_add_console_app(field_preset_tests PRODUCT_NAME "Preset format tests")
    target_sources(field_preset_tests PRIVATE tests/preset_tests.cpp Source/PresetFormat.cpp Source/PresetMapping.cpp Source/PresetParameters.cpp)
    target_include_directories(field_preset_tests PRIVATE Source)
    target_compile_definitions(field_preset_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
    target_link_libraries(field_preset_tests PRIVATE juce::juce_core)
    add_test(NAME preset COMMAND field_preset_tests)
    juce_add_console_app(field_parameter_tests PRODUCT_NAME "Preset parameter tests")
    target_sources(field_parameter_tests PRIVATE tests/preset_parameter_tests.cpp Source/PresetParameters.cpp)
    target_include_directories(field_parameter_tests PRIVATE Source)
    target_compile_definitions(field_parameter_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
    target_link_libraries(field_parameter_tests PRIVATE juce::juce_core)
    add_test(NAME parameters COMMAND field_parameter_tests)
    if(UNIX AND NOT APPLE)
      add_executable(field_vst3_host tests/vst3_host.cpp)
      target_include_directories(field_vst3_host PRIVATE "${FIELD_JUCE_ROOT}/modules/juce_audio_processors_headless/format_types/VST3_SDK")
      target_link_libraries(field_vst3_host PRIVATE ${CMAKE_DL_LIBS})
      add_test(NAME vst3_host COMMAND field_vst3_host $<TARGET_FILE:FieldStudio_VST3>)
    endif()
  endif()
endif()
